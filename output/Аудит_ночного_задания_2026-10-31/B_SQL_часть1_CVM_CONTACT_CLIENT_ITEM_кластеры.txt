П1. Фактическое время и сбросы каждого оператора процедур части 1 (кэш планов; номер строки ±1 к файлам audit\procs):
    SELECT proc_name = OBJECT_NAME(st.objectid, st.dbid)
         , line_no   = LEN(LEFT(st.text, qs.statement_start_offset / 2))
                       - LEN(REPLACE(LEFT(st.text, qs.statement_start_offset / 2), CHAR(10), '')) + 1
         , stmt      = LEFT(SUBSTRING(st.text, qs.statement_start_offset / 2 + 1, 4000), 150)
         , qs.execution_count
         , last_sec  = qs.last_elapsed_time / 1000000
         , avg_sec   = qs.total_elapsed_time / 1000000 / NULLIF(qs.execution_count, 0)
         , last_cpu_sec = qs.last_worker_time / 1000000
         , qs.last_rows, last_grant_mb = qs.last_grant_kb / 1024, last_used_mb = qs.last_used_grant_kb / 1024
         , qs.last_spills, qs.last_dop, qs.creation_time, qs.last_execution_time
    FROM sys.dm_exec_query_stats AS qs
    CROSS APPLY sys.dm_exec_sql_text(qs.sql_handle) AS st
    WHERE st.dbid = DB_ID(N'mci_model')
      AND OBJECT_NAME(st.objectid, st.dbid) IN (N'CVM_CONTACT', N'CVM_CONTACT_DATA_W', N'CVM_CONTACT_DATA', N'client_item_load',
          N'client_item_keys_day', N'client_item_stats', N'clusters', N'cluster_assign', N'cluster_assign_calc',
          N'cluster_score', N'assort_week', N'cont_price_segment', N'product_score_region')
    ORDER BY proc_name, qs.last_elapsed_time DESC;
    -- и включён ли Query Store (история по ночам):
    SELECT actual_state_desc, max_storage_size_mb FROM sys.database_query_store_options;

П2. Пошаговое время clusters и присвоения за 2 ночи (журналы витрины):

         , SEC = DATEDIFF(second, STARTED, FINISHED)
