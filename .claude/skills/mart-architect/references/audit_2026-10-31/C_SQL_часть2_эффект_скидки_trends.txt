-- 1. Длительность шагов по датам за 60 дней (видно дни 1–10, первую пятницу, воскресенья)
SELECT h.step_id, h.step_name, h.run_date,
       dur_min = CAST((h.run_duration / 10000 * 3600 + h.run_duration / 100 % 100 * 60 + h.run_duration % 100) / 60.0 AS decimal(9, 1))
FROM msdb.dbo.sysjobhistory AS h WITH (NOLOCK)
JOIN msdb.dbo.sysjobs AS j WITH (NOLOCK) ON j.job_id = h.job_id
WHERE j.name = N'MCI Regular update' AND h.step_id IN (27, 29, 40, 46, 48, 49, 50, 51, 52, 53, 54, 55, 56)
  AND h.run_date >= CAST(CONVERT(char(8), DATEADD(day, -60, GETDATE()), 112) AS int)
ORDER BY h.step_id, h.run_date;

-- 2. Вывод шага 52 (порции, акции, время чтения чеков, остановка по 18:00) за 5 последних запусков
SELECT TOP (5) h.run_date, h.run_time, h.message
FROM msdb.dbo.sysjobhistory AS h WITH (NOLOCK)
JOIN msdb.dbo.sysjobs AS j WITH (NOLOCK) ON j.job_id = h.job_id
WHERE j.name = N'MCI Regular update' AND h.step_id = 52
ORDER BY h.instance_id DESC;

-- 3. Журнал модели скидок: окно, дней, самая долгая порция, реестр, цены зон, регулярная цена

FROM dbo.I_DISC_MODEL_LOG WITH (NOLOCK) ORDER BY RUN_ID DESC;

-- 4. Витрина скидок: дни «без начислений» по кругу и журнал

SELECT TOP (10) * FROM dbo.I_DISCOUNT_LOG WITH (NOLOCK) ORDER BY RUN_ID DESC;

-- 7. I_PROMO_OFFER: упорядоченность групп строк по ID_PROMO (для E3) и доля удалённых строк в больших CCI
SELECT s.segment_id, s.min_data_id, s.max_data_id, s.row_count
FROM sys.column_store_segments AS s WITH (NOLOCK)
JOIN sys.partitions AS p WITH (NOLOCK) ON p.hobt_id = s.hobt_id
JOIN sys.columns AS c WITH (NOLOCK) ON c.object_id = p.object_id AND c.column_id = s.column_id
WHERE p.object_id = OBJECT_ID(N'dbo.I_PROMO_OFFER') AND c.name = N'ID_PROMO'
ORDER BY s.segment_id DESC
OFFSET 0 ROWS FETCH NEXT 200 ROWS ONLY;
SELECT OBJECT_NAME(rg.object_id) AS tbl, rg.state_desc, groups = count(*), rows_total = sum(rg.total_rows),
       rows_deleted = sum(rg.deleted_rows), small_groups = sum(CASE WHEN rg.total_rows < 102400 THEN 1 ELSE 0 END)
FROM sys.column_store_row_groups AS rg WITH (NOLOCK)
WHERE rg.object_id IN (OBJECT_ID(N'dbo.I_DISC_MODEL_CLIENT_DISC'), OBJECT_ID(N'dbo.I_DISC_MODEL_CLIENT_DAY'),
                       OBJECT_ID(N'dbo.I_DISC_MODEL_ZONE_REG'), OBJECT_ID(N'dbo.I_DISCOUNT_CHECK'),
                       OBJECT_ID(N'dbo.I_BONUS_CLIENT_MONTH'), OBJECT_ID(N'dbo.I_PROMO_OFFER'))
GROUP BY rg.object_id, rg.state_desc;

-- 8. Колонки I_CH_INDEXES (есть ли DATA / YEAR_MONTH для фильтра в T2) и откуда берутся PENETRATION_CHECK, PRICE_INDEX,
--    BONUS_PAY, BONUS_ACCRUAL; есть ли REFERENCE_DATE в I_CVM_CONTACT (S2)
SELECT OBJECT_NAME(c.object_id) AS tbl, c.name, TYPE_NAME(c.user_type_id) AS type_name

WHERE c.object_id IN (OBJECT_ID(N'dbo.I_CH_INDEXES'), OBJECT_ID(N'dbo.I_CHECKHEADER'), OBJECT_ID(N'dbo.I_CVM_CONTACT'),
                      OBJECT_ID(N'dbo.I_CHECK'))

                 N'COST', N'QUANTITY', N'COST_DISCOUNT', N'DISCOUNT', N'CARD_DISCOUNT', N'REAL_COST')
ORDER BY tbl, c.name;

SELECT c.name, TYPE_NAME(c.user_type_id) AS type_name, c.precision, c.scale

WHERE c.object_id = OBJECT_ID(N'dbo.I_CHECK') AND c.name IN (N'COST', N'QUANTITY', N'COST_DISCOUNT', N'DISCOUNT', N'CARD_DISCOUNT', N'REAL_COST');

-- 9. Акции эффекта: очередь ночи (нет в витрине при старте за 120 дней) и сколько акций считается за ночь
SELECT n_missing = count(DISTINCT p.ID_PROMO)
FROM dbo.I_PROMO AS p WITH (NOLOCK)
WHERE p.ID_COMPANY = 1 AND p.ID_PROMO <> 1013000
  AND (p.ID_PROMO BETWEEN 101000 AND 101999 OR p.ID_PROMO BETWEEN 110000 AND 111999
       OR (p.ID_PROMO BETWEEN 500000 AND 599999 AND p.TYPE_PROMO = N'CVM'))
  AND p.START_DATE >= DATEADD(day, -120, CAST(GETDATE() AS date)) AND p.START_DATE < CAST(GETDATE() AS date)
  AND NOT EXISTS (SELECT 1 FROM dbo.I_CVM_EFFECT AS e WITH (NOLOCK) WHERE e.ID_COMPANY = 1 AND e.[ID] = p.ID_PROMO);
SELECT calc_day = CAST(e.[Дата расчета] AS date), promos = count(DISTINCT e.[ID])
FROM dbo.I_CVM_EFFECT AS e WITH (NOLOCK)
WHERE e.ID_COMPANY = 1
GROUP BY CAST(e.[Дата расчета] AS date)
ORDER BY calc_day DESC;
